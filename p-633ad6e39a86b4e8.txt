+// The list behind This week at home (Emjee 23 nov), kept apart from the screen so it can be tested.
+import { myRooms, planCandidates } from "./my-rooms";
+import { canDo, DAY, isManual, nextReady, snoozed, startOfTomorrow, type CareState } from "./rules";
+
+export type WeekTask = { id: string; name: string; room: string; at: number; diff: number };
+
+const dayStart = (t: number) => {
+  const d = new Date(t);
+  d.setHours(0, 0, 0, 0);
+  return d.getTime();
+};
+
+export function weekTasks(s: CareState, now: number): WeekTask[] {
+  const roomName = new Map(myRooms(s).map((r) => [r.key, r.name]));
+  const out: WeekTask[] = [];
+  for (const c of planCandidates(s)) {
+    if (isManual(c.days) || !s.last[c.id]) continue;
+    let at = canDo(s, c.id, c.days, now) ? now : nextReady(s, c.id, c.days);
+    if (at == null) continue;
+    if (snoozed(s, c.id, now) && at <= now) at = startOfTomorrow(now);
+    const diff = Math.round((dayStart(at) - dayStart(now)) / DAY);
+    if (diff > 6) continue;
+    out.push({ id: c.id, name: c.task.name, room: roomName.get(c.room) ?? c.room, at, diff: Math.max(0, diff) });
+  }
+  return out.sort((a, b) => a.diff - b.diff || a.room.localeCompare(b.room) || a.name.localeCompare(b.name));
+}
+
+export const whenText = (diff: number) => (diff <= 0 ? "today" : diff === 1 ? "tomorrow" : `in ${diff} days`);
